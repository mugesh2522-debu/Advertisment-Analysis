import pickle
import numpy as np
import pandas as pd
from fastapi import FastAPI
from fastapi.middleware.cors import CORSMiddleware
from pydantic import BaseModel

# 1. Load your serialized scikit-learn 1.6.1 model
with open("Linear regression.pkl", "rb") as f:
    model = pickle.load(f)

app = FastAPI(title="Marketing Mix Linear Regression API")

# Enable CORS for frontend integration
app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_methods=["*"],
    allow_headers=["*"],
)

class AdSpendRequest(BaseModel):
    TV: float
    Radio: float
    Newspaper: float

@app.post("/predict")
def predict_sales(spend: AdSpendRequest):
    # Maintain exact feature order: ['TV', 'Radio', 'Newspaper']
    features = pd.DataFrame([{
        "TV": spend.TV,
        "Radio": spend.Radio,
        "Newspaper": spend.Newspaper
    }])
    
    prediction = float(model.predict(features)[0])
    
    return {
        "predicted_sales": round(prediction, 4),
        "intercept": float(model.intercept_),
        "coefficients": dict(zip(model.feature_names_in_, model.coef_.tolist()))
    }

# Run with: uvicorn app:app --reload
